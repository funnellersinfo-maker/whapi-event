    54→function useNow(intervalMs = 1000) {
    55→  const [now, setNow] = useState<Date | null>(null);
    56→  useEffect(() => {
    57→    // Primer valor diferido al siguiente tick (evita cascadas en el mount)
    58→    const boot = setTimeout(() => setNow(new Date()), 0);
    59→    const id = setInterval(() => setNow(new Date()), intervalMs);
    60→    return () => {
    61→      clearTimeout(boot);
    62→      clearInterval(id);
    63→    };
    64→  }, [intervalMs]);
    65→  return now;
    66→}

    68→function formatCountdown(ms: number): string {
    69→  const total = Math.floor(ms / 1000);
    70→  const h = Math.floor(total / 3600);
    71→  const m = Math.floor((total % 3600) / 60);
    72→  const s = total % 60;
    73→  return [h, m, s].map((v) => String(v).padStart(2, "0")).join(":");
    74→}

   525→
   526→function localDateKey(): string {
   527→  const d = new Date();
   528→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
   529→}
   530→
   531→function readBookingsToday(board: DayBoard): RegistrationData | null {
   532→  try {
   533→    const raw = window.localStorage.getItem("wa_event_bookings_v1");
   534→    if (!raw) return null;
   535→    const list = JSON.parse(raw) as Array<{
   536→      slot: string;
   537→      dateKey: string;
   538→      name: string;
   539→      city: string;
   540→      business: string;
   541→    }>;
   542→    const today = localDateKey();
   543→    const found = [...list].reverse().find((b) => b.dateKey === today);
   544→    if (!found) return null;
   545→    return { ...found, day: "hoy" };
   546→  } catch {
   547→    return null;
   548→  }
   549→}

   600→function SlotCard({
   601→  slot,
   602→  index,
   603→  onSelect,
   604→}: {
   605→  slot: SlotState;
   606→  index: number;
   607→  onSelect: (slot: SlotState) => void;
   608→}) {
   609→  const status = slot.passed
   610→    ? { label: "COMENZÓ", tone: "passed" as const }
   611→    : slot.seatsLeft === 0
   612→      ? { label: "AGOTADO", tone: "full" as const }
   613→      : slot.seatsLeft === 1
   614→        ? { label: "1 CUPO", tone: "scarce" as const }
   615→        : { label: `${slot.seatsLeft} CUPOS`, tone: "open" as const };
   616→
   617→  const disabled = !slot.available;
   618→
   619→  const base =
   620→    "group relative flex flex-col items-center gap-1.5 rounded-2xl px-3 py-4 text-center transition-all duration-300";
   621→  const toneClass =
   622→    status.tone === "passed"
   623→      ? "border-white/5 bg-white/[0.02] opacity-45"
   624→      : status.tone === "full"
   625→        ? "border-white/6 bg-white/[0.02] opacity-55"
   626→        : "glass card-lift cursor-pointer hover:border-wa/45";
   627→
   628→  return (
   629→    <motion.button
   630→      type="button"
   631→      initial={{ opacity: 0, y: 18 }}
   632→      whileInView={{ opacity: 1, y: 0 }}
   633→      viewport={{ once: true }}
   634→      transition={{ delay: index * 0.05, duration: 0.45 }}
   635→      onClick={() => onSelect(slot)}
   636→      disabled={disabled}
   637→      aria-label={
   638→        disabled
   639→          ? `Sesión ${slot.label} no disponible`
   640→          : slot.seatsLeft === 1
   641→            ? `Reservar sesión de hoy ${slot.label}, queda 1 cupo`
   642→            : `Reservar sesión de hoy ${slot.label}, ${slot.seatsLeft} cupos disponibles`
   643→      }
   644→      aria-disabled={disabled}
   645→      className={`${base} ${toneClass} ${disabled ? "cursor-not-allowed" : ""}`}
   646→    >
   647→      {status.tone === "scarce" && (
   648→        <Flame
   649→          className="absolute right-2.5 top-2.5 h-3.5 w-3.5 text-amber-warn"
   650→          aria-hidden="true"
   651→        />
   652→      )}
   653→      {status.tone === "passed" && (
   654→        <Lock className="absolute right-2.5 top-2.5 h-3.5 w-3.5 text-white/30" aria-hidden="true" />
   655→      )}
   656→
   657→      <span
   658→        className={`font-mono text-lg font-bold tabular-nums ${
   659→          disabled ? "text-white/35 line-through" : "text-white"
   660→        }`}
   661→      >
   662→        {slot.label}
   663→      </span>
   664→      <span
   665→        className={`text-[10px] font-bold uppercase tracking-[0.14em] ${
   666→          status.tone === "scarce"
   667→            ? "text-amber-warn"
   668→            : status.tone === "full" || status.tone === "passed"
   669→              ? "text-white/40"
   670→              : "text-wa/90"
   671→        }`}
   672→      >
   673→        {status.label}
   674→      </span>
   675→    </motion.button>
   676→  );
   677→}
