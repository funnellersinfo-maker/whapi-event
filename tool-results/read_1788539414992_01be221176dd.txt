    54→    54→function useNow(intervalMs = 1000) {
    55→    55→  const [now, setNow] = useState<Date | null>(null);
    56→    56→  useEffect(() => {
    57→    57→    // Primer valor diferido al siguiente tick (evita cascadas en el mount)
    58→    58→    const boot = setTimeout(() => setNow(new Date()), 0);
    59→    59→    const id = setInterval(() => setNow(new Date()), intervalMs);
    60→    60→    return () => {
    61→    61→      clearTimeout(boot);
    62→    62→      clearInterval(id);
    63→    63→    };
    64→    64→  }, [intervalMs]);
    65→    65→  return now;
    66→    66→}
    67→    67→
    68→    68→function formatCountdown(ms: number): string {
    69→    69→  const total = Math.floor(ms / 1000);
    70→    70→  const h = Math.floor(total / 3600);
    71→    71→  const m = Math.floor((total % 3600) / 60);
    72→    72→  const s = total % 60;
    73→    73→  return [h, m, s].map((v) => String(v).padStart(2, "0")).join(":");
    74→    74→}

   600→   600→function SlotCard({
   601→   601→  slot,
   602→   602→  index,
   603→   603→  onSelect,
   604→   604→}: {
   605→   605→  slot: SlotState;
   606→   606→  index: number;
   607→   607→  onSelect: (slot: SlotState) => void;
   608→   608→}) {
   609→   609→  const status = slot.passed
   610→   610→    ? { label: "COMENZÓ", tone: "passed" as const }
   611→   611→    : slot.seatsLeft === 0
   612→   612→      ? { label: "AGOTADO", tone: "full" as const }
   613→   613→      : slot.seatsLeft === 1
   614→   614→        ? { label: "1 CUPO", tone: "scarce" as const }
   615→   615→        : { label: `${slot.seatsLeft} CUPOS`, tone: "open" as const };
   616→   616→
   617→   617→  const disabled = !slot.available;
   618→   618→
   619→   619→  const base =
   620→   620→    "group relative flex flex-col items-center gap-1.5 rounded-2xl px-3 py-4 text-center transition-all duration-300";
   621→   621→  const toneClass =
   622→   622→    status.tone === "passed"
   623→   623→      ? "border-white/5 bg-white/[0.02] opacity-45"
   624→   624→      : status.tone === "full"
   625→   625→        ? "border-white/6 bg-white/[0.02] opacity-55"
   626→   626→        : "glass card-lift cursor-pointer hover:border-wa/45";
   627→   627→
   628→   628→  return (
   629→   629→    <motion.button
   630→   630→      type="button"
   631→   631→      initial={{ opacity: 0, y: 18 }}
   632→   632→      whileInView={{ opacity: 1, y: 0 }}
   633→   633→      viewport={{ once: true }}
   634→   634→      transition={{ delay: index * 0.05, duration: 0.45 }}
   635→   635→      onClick={() => onSelect(slot)}
   636→   636→      disabled={disabled}
   637→   637→      aria-label={
   638→   638→        disabled
   639→   639→          ? `Sesión ${slot.label} no disponible`
   640→   640→          : slot.seatsLeft === 1
   641→   641→            ? `Reservar sesión de hoy ${slot.label}, queda 1 cupo`
   642→   642→            : `Reservar sesión de hoy ${slot.label}, ${slot.seatsLeft} cupos disponibles`
   643→   643→      }
   644→   644→      aria-disabled={disabled}
   645→   645→      className={`${base} ${toneClass} ${disabled ? "cursor-not-allowed" : ""}`}
   646→   646→    >
   647→   647→      {status.tone === "scarce" && (
   648→   648→        <Flame
   649→   649→          className="absolute right-2.5 top-2.5 h-3.5 w-3.5 text-amber-warn"
   650→   650→          aria-hidden="true"
   651→   651→        />
   652→   652→      )}
   653→   653→      {status.tone === "passed" && (
   654→   654→        <Lock className="absolute right-2.5 top-2.5 h-3.5 w-3.5 text-white/30" aria-hidden="true" />
   655→   655→      )}
   656→   656→
   657→   657→      <span
   658→   658→        className={`font-mono text-lg font-bold tabular-nums ${
   659→   659→          disabled ? "text-white/35 line-through" : "text-white"
   660→   660→        }`}
   661→   661→      >
   662→   662→        {slot.label}
   663→   663→      </span>
   664→   664→      <span
   665→   665→        className={`text-[10px] font-bold uppercase tracking-[0.14em] ${
   666→   666→          status.tone === "scarce"
   667→   667→            ? "text-amber-warn"
   668→   668→            : status.tone === "full" || status.tone === "passed"
   669→   669→              ? "text-white/40"
   670→   670→              : "text-wa/90"
   671→   671→        }`}
   672→   672→      >
   673→   673→        {status.label}
   674→   674→      </span>
   675→   675→    </motion.button>
   676→   676→  );
   677→   677→}
